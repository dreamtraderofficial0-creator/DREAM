      function generateDreamReferralCode(existingUsers) {
        const used = new Set((existingUsers || []).map((u) => (u.referralCode || '').toUpperCase()));
        let code = '';
        let attempts = 0;
        do {
          const digits = Math.floor(1000 + Math.random() * 9000);
          code = 'DREAM-' + digits;
          attempts += 1;
        } while (used.has(code) && attempts < 50);
        return code;
      }

            ...JSON.parse(JSON.stringify(defaultState)),

            transactions: Array.isArray(parsed.transactions) ? parsed.transactions : []

          return JSON.parse(JSON.stringify(defaultState));
        }
      }

          warning: 'bg-navy-900 border-gold-500/50 text-gold-300',
          error: 'bg-navy-900 border-rose-500/50 text-rose-300'
        };

        toast.className = `pointer-events-auto rounded-xl border px-4 py-3 text-xs font-semibold shadow-xl ${palette[type] || palette.success}`;
        toast.textContent = message;
        container.appendChild(toast);
        setTimeout(() => {
          toast.remove();

      function getUserDailyProfitRate(user) {
        if (!user) return 0;
        const activeList = Array.isArray(user.activeInvestments) ? user.activeInvestments : [];
        if (activeList.length > 0) {
          return activeList.reduce((sum, inv) => {
            if (inv.status === 'Active') {
              return sum + Math.round((Number(inv.amount || 0) * Number(inv.dailyReturn || 0)) / 100);
            }
            return sum;
          }, 0);
        }
        if (Number(user.investedAmount || 0) > 0 && user.activePlanName) {
          const plan = state.plans.find((p) => p.name === user.activePlanName);
          const roi = plan ? Number(plan.dailyReturn || 0) : 0;
          return Math.round((Number(user.investedAmount || 0) * roi) / 100);
        }
        return 0;
      }

      document.getElementById('withdrawForm').addEventListener('submit', function (e) {
        e.preventDefault();
        const user = getCurrentUser();
        if (!user) {
          showToast('Please register or log in first', 'warning');
          return;
        }

        const jazzCashNumber = document.getElementById('withdrawJazzCashNumber').value.trim();
        const amount = Number(document.getElementById('withdrawAmount').value);
        const minWd = Number(state.settings.minWithdrawal || 500);

        if (!jazzCashNumber) {
          showToast('Please enter your JazzCash withdrawal number', 'error');
          return;
        }
        if (!amount || amount < minWd) {
          showToast('Minimum withdrawal amount is ' + formatPKR(minWd), 'error');
          return;
        }
        if (amount > Number(user.balance || 0)) {
          showToast('Insufficient balance for this withdrawal', 'error');
          return;
        }

        // Hold requested amount in pending withdrawal until admin approves or rejects
        user.balance = Number(user.balance || 0) - amount;

        state.transactions.unshift({
          id: Date.now(),
          userId: user.id,
          userName: user.name,
          type: 'Withdrawal',
          amount,
          account: jazzCashNumber,
          status: 'Pending',
          time: 'Just now'
        });

        saveState();
        this.reset();
        renderUserPortal();
        showToast('Withdrawal request submitted (Pending admin approval)', 'success');
      });

        const hash = (window.location.hash || '').toLowerCase();

      window.creditUserDailyProfit = function (userId) {
        const user = state.users.find((u) => u.id === userId);
        if (!user) return;
        const profitAmount = getUserDailyProfitRate(user);
        if (profitAmount <= 0) {
          showToast('No daily profit rate available for ' + user.name, 'warning');
          return;
        }

        // Approving daily profit updates both Approved Profit and Current Balance
        user.approvedProfit = Number(user.approvedProfit || 0) + profitAmount;
        user.balance = Number(user.balance || 0) + profitAmount;

        state.transactions.unshift({
          id: Date.now() + Math.floor(Math.random() * 100),
          userId: user.id,
          userName: user.name,
          type: 'Daily Profit',
          amount: profitAmount,
          details: (user.activePlanName || 'Investment') + ' Daily Profit',
          status: 'Approved',
          time: 'Just now'
        });

        saveState();
        renderAdminControlPanel();
        renderUserPortal();
        showToast(`Approved ${formatPKR(profitAmount)} profit for ${user.name}`, 'success');
      };

      window.approveAllDailyProfits = function () {
        let count = 0;
        let total = 0;

        state.users.forEach((user) => {
          const profitAmount = getUserDailyProfitRate(user);
          if (profitAmount > 0) {
            user.approvedProfit = Number(user.approvedProfit || 0) + profitAmount;
            user.balance = Number(user.balance || 0) + profitAmount;
            count += 1;
            total += profitAmount;

            state.transactions.unshift({
              id: Date.now() + count,
              userId: user.id,
              userName: user.name,
              type: 'Daily Profit',
              amount: profitAmount,
              details: (user.activePlanName || 'Investment') + ' Daily Profit',
              status: 'Approved',
              time: 'Just now'
            });
          }
        });

        if (count === 0) {
          showToast('No active investments found for daily profit payout', 'warning');
          return;
        }

        saveState();
        renderAdminControlPanel();
        renderUserPortal();
        showToast(`Approved ${count} daily profit payouts (${formatPKR(total)})`, 'success');
      };

      window.reviewTransaction = function (txId, decision) {
        const tx = state.transactions.find((t) => t.id === txId);
        if (!tx || tx.status !== 'Pending') return;
        tx.status = decision;
        tx.time = 'Just now';

        const user = state.users.find((u) => u.id === tx.userId);
        if (tx.type === 'Deposit' && decision === 'Approved' && user) {
          const amt = Number(tx.amount || 0);
          const matchedPlan = state.plans.find((p) => p.name === tx.planName) || state.plans[0];
          user.investedAmount = Number(user.investedAmount || 0) + amt;
          user.activePlanName = matchedPlan ? matchedPlan.name : (tx.planName || 'Starter');

          if (!Array.isArray(user.activeInvestments)) user.activeInvestments = [];
          user.activeInvestments.push({
            id: Date.now(),
            planName: user.activePlanName,
            amount: amt,
            dailyReturn: matchedPlan ? matchedPlan.dailyReturn : 8,
            duration: matchedPlan ? matchedPlan.duration : 15,
            status: 'Active'
          });

          if (user.referredBy) {
            const sponsor = state.users.find((u) => (u.referralCode || '').toUpperCase() === user.referredBy.toUpperCase());
            if (sponsor) {
              const bonus = Math.round((amt * Number(state.settings.referralBonusPct || 10)) / 100);
              sponsor.balance = Number(sponsor.balance || 0) + bonus;
              sponsor.referralBonusEarned = Number(sponsor.referralBonusEarned || 0) + bonus;
            }
          }
        }

        if (tx.type === 'Withdrawal' && decision === 'Rejected' && user) {
          user.balance = Number(user.balance || 0) + Number(tx.amount || 0);
        }

        saveState();
        renderAdminControlPanel();
        renderUserPortal();
        showToast(`${tx.type} marked as ${decision}`, decision === 'Approved' ? 'success' : 'warning');
      };
